const GROQ_ENDPOINT = "https://api.groq.com/openai/v1/chat/completions";
const ALLOWED_ORIGIN = "*";

// ============================================================
// CORS & RESPONSE HELPERS
// ============================================================

function corsHeaders() {
    return {
        "Access-Control-Allow-Origin": ALLOWED_ORIGIN,
        "Access-Control-Allow-Methods": "POST, OPTIONS",
        "Access-Control-Allow-Headers": "Content-Type"
    };
}

function jsonResponse(body, status = 200) {
    return new Response(
        JSON.stringify(body),
        {
            status,
            headers: {
                "Content-Type": "application/json",
                ...corsHeaders()
            }
        }
    );
}

// ============================================================
// ROBUST JSON VALIDATION & REPAIR
// ============================================================

function extractAndRepairJSON(text) {
    if (!text) return null;

    let cleaned = text
        .replace(/```json/gi, "")
        .replace(/```/g, "")
        .trim();

    const start = cleaned.indexOf("{");
    const end = cleaned.lastIndexOf("}");

    if (start === -1 || end === -1 || end < start) {
        return null;
    }

    const jsonStr = cleaned.slice(start, end + 1);

    try {
        return JSON.parse(jsonStr);
    } catch (e) {
        const repaired = jsonStr
            .replace(/,\s*}/g, "}")
            .replace(/,\s*]/g, "]");

        try {
            return JSON.parse(repaired);
        } catch {
            return null;
        }
    }
}

function clampValue(value, min, max, defaultValue) {
    const num = Number(value);
    if (!Number.isFinite(num)) return defaultValue;
    return Math.max(min, Math.min(max, num));
}

function ensureArray(value, defaultValue = []) {
    if (Array.isArray(value) && value.length > 0) {
        return value;
    }
    return defaultValue;
}

function ensureString(value, defaultValue = "") {
    const str = String(value || "").trim();
    return str || defaultValue;
}

function ensureEnum(value, allowed, defaultValue) {
    const normalized = String(value || "").trim().toLowerCase();
    return allowed.includes(normalized) ? normalized : defaultValue;
}

/**
 * jsonMode è ora un parametro esplicito, non più sempre true.
 * - true  → usato SOLO da handleHookAnalysis/handlePersonalCoach,
 *           che dichiarano uno schema a oggetto {} nel prompt.
 * - false → route generica (js_api.js): idee separate da "|",
 *           array JSON nudi [...], testo libero. Forzare json_object
 *           qui rompeva/rifiutava sistematicamente l'output di Groq.
 */
async function callGroq(messages, env, temperature = 0.2, maxTokens = 3000, jsonMode = false) {
    if (!env.CHIAVE_API_AI) {
        throw new Error("Missing AI key");
    }

    const body = {
        model: "llama-3.3-70b-versatile",
        messages,
        temperature,
        max_tokens: maxTokens
    };

    if (jsonMode) {
        body.response_format = { type: "json_object" };
    }

    const response = await fetch(GROQ_ENDPOINT, {
        method: "POST",
        headers: {
            "Content-Type": "application/json",
            "Authorization": `Bearer ${env.CHIAVE_API_AI}`
        },
        body: JSON.stringify(body)
    });

    if (!response.ok) {
        const errorText = await response.text();
        throw new Error(`Groq API Error: ${response.status} - ${errorText}`);
    }

    const data = await response.json();
    return data;
}

{
  "hookScore": 0-100,

  "originalityScore": 0-100,
  "clarityScore": 0-100,
  "emotionalTrigger": "brief description",
  "viewerCuriosity": "high/medium/low",
  "expectedRetention": "high/medium/low",
  "predictedFirstImpression": "brief description",
  "strengths": ["strength 1", "strength 2"],
  "weaknesses": ["weakness 1", "weakness 2"],
  "missedOpportunities": ["opportunity 1", "opportunity 2"],
  "recommendedImprovements": ["improvement 1", "improvement 2"],

  "improvedHookVersionB": "aggressive improvement",
  "improvedHookVersionC": "viral improvement",
  "finalSummary": "brief summary"
}`;

async function handleHookAnalysis(payload, env) {
    const { concept, channelProfile } = payload;

    if (!concept || typeof concept !== "string") {
        return { error: "Missing concept description" };
    }

    let context = "";
    if (channelProfile) {
        const avgViews = channelProfile.averageViews || 0;
        const bestFormats = Array.isArray(channelProfile.bestFormats)
            ? channelProfile.bestFormats.slice(0, 3).join(", ")
            : "None";
        const worstFormats = Array.isArray(channelProfile.worstFormats)
            ? channelProfile.worstFormats.slice(0, 3).join(", ")
            : "None";

        context = `
CHANNEL CONTEXT:
- Average views: ${avgViews}
- Best performing formats: ${bestFormats}
- Underperforming formats: ${worstFormats}

Use this context to personalize your analysis. Reference historical performance when relevant.
`;
    }

    const userPrompt = `${context}

OPENING CONCEPT TO ANALYZE:
"${concept}"

Analyze this opening concept following the process above. Consider how it aligns with the channel's historical performance if available.`;

    try {
        const response = await callGroq(
            [
                { role: "system", content: HOOK_ANALYZER_PROMPT },
                { role: "user", content: userPrompt }
            ],
            env,
            0.18,
            3000,
            true // jsonMode: questo handler dichiara davvero uno schema a oggetto
        );

        const content = response.choices?.[0]?.message?.content || "";
        const parsed = extractAndRepairJSON(content);

        if (!parsed) {
            return { error: "AI returned invalid JSON" };
        }

        return {
            hookScore: clampValue(parsed.hookScore, 0, 100, 50),
            curiosityScore: clampValue(parsed.curiosityScore, 0, 100, 50),
            scrollStopScore: clampValue(parsed.scrollStopScore, 0, 100, 50),
            originalityScore: clampValue(parsed.originalityScore, 0, 100, 50),
            clarityScore: clampValue(parsed.clarityScore, 0, 100, 50),
            emotionalTrigger: ensureString(parsed.emotionalTrigger, "Not detected"),
            viewerCuriosity: ensureString(parsed.viewerCuriosity, "medium"),
            expectedRetention: ensureString(parsed.expectedRetention, "medium"),
            predictedFirstImpression: ensureString(parsed.predictedFirstImpression, "Unknown"),
            strengths: ensureArray(parsed.strengths, ["No specific strengths detected"]),
            weaknesses: ensureArray(parsed.weaknesses, ["No specific weaknesses detected"]),
            missedOpportunities: ensureArray(parsed.missedOpportunities, ["No missed opportunities identified"]),
            recommendedImprovements: ensureArray(parsed.recommendedImprovements, ["No specific recommendations"]),
            improvedHookVersionA: ensureString(parsed.improvedHookVersionA, concept),
            improvedHookVersionB: ensureString(parsed.improvedHookVersionB, concept),
            improvedHookVersionC: ensureString(parsed.improvedHookVersionC, concept),
            finalSummary: ensureString(parsed.finalSummary, "Analysis complete")
        };

    } catch (error) {
        return { error: error.message };
    }
}

// AI ROUTER - MAIN DISPATCHER
// ============================================================

export default {
    async fetch(request, env) {
        if (request.method === "OPTIONS") {
            return new Response(null, { headers: corsHeaders() });
        }

        if (request.method !== "POST") {
            return jsonResponse({ error: "Method not allowed" }, 405);
        }

        if (!env.CHIAVE_API_AI) {
            return jsonResponse({ error: "Missing AI key" }, 500);
        }

        let payload;
        try {
            payload = await request.json();
        } catch {
            return jsonResponse({ error: "Invalid JSON" }, 400);
        }

        const { type } = payload;

        try {
            // =============================
            // ROUTE: HOOK & CONCEPT ANALYZER
            // =============================
            if (type === "hook_analysis") {
                const result = await handleHookAnalysis(payload, env);
                return jsonResponse(result);
            }

            // =============================
            // ROUTE: PERSONAL AI COACH
            // =============================
            if (type === "personal_coach") {
                const result = await handlePersonalCoach(payload, env);
                return jsonResponse(result);
            }

            // =============================
            // ROUTE: TITLE OPTIMIZER
            // =============================
            if (type === "title_optimizer") {
                const result = await handleTitleOptimizer(payload, env);
                return jsonResponse(result);
            }


            // =============================
            // ROUTE: GENERIC OPENAI COMPATIBLE (fallback)
            // Usata da js_api.js per: generazione idee (testo con "|"),
            // discoverChannelFormats/expandFormatCoverage (array JSON
            // nudo), generateKeywordsForFormat (array JSON nudo).
            // NON deve mai forzare json_object: nessuno di questi
            // payload è un oggetto {} — forzarlo fa scartare/rompere
            // la risposta da Groq quasi sistematicamente.
            // =============================
            const { messages, model, temperature, max_tokens } = payload;

            if (!Array.isArray(messages) || messages.length === 0) {
                return jsonResponse({ error: "Missing messages" }, 400);
            }

            const data = await callGroq(
                messages,
                env,
                temperature ?? 0.7,
                max_tokens ?? 2000,
                false // jsonMode disattivato: vedi nota sopra
            );

            return jsonResponse(data);

        } catch (error) {
            return jsonResponse({ error: error.message }, 500);
        }
    }
};
