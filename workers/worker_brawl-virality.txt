const ALLOWED_ORIGIN = [
    "https://brawl-analystic-official.vercel.app",
    "https://www.brawl-analystic-official.vercel.app",
    "http://localhost:5500",
    "http://127.0.0.1:5500"
];

function corsHeaders(origin) {

    const allowed =
        origin === ALLOWED_ORIGIN;

    return {

        "Access-Control-Allow-Origin":
            allowed ? origin : ALLOWED_ORIGIN,

        "Access-Control-Allow-Methods":
            "POST, OPTIONS",

        "Access-Control-Allow-Headers":
            "Content-Type"

    };

}

function json(body, status = 200, origin = ALLOWED_ORIGIN) {

    return new Response(

        JSON.stringify(body, null, 2),

        {

            status,

            headers: {

                "Content-Type":
                    "application/json",

                ...corsHeaders(origin)

            }

        }

    );

}

async function getDailyTrends(env) {

    const fallbackTrends = [

        {

            query:
                "brawl stars meta tier list"

        },

        {

            query:
                "ranked mode global leaderboard"

        },

        {

            query:
                "best brawlers hypercharge"

        }

    ];

    try {

        const response =
            await fetch(

                "https://trends.angeskicollab10.workers.dev/trends"

            );

        if (!response.ok) {

            return {

                trends:
                    fallbackTrends

            };

        }

        const data =
            await response.json();

        if (

            !Array.isArray(data.trends)

            ||

            data.trends.length === 0

        ) {

            return {

                trends:
                    fallbackTrends

            };

        }

        return data;

    }

    catch {

        return {

            trends:
                fallbackTrends

        };

    }

}

function buildPrompt(

    proposal,

    channelProfile,

    trends

) {

    const listaTrends =

        (trends?.trends || [])

            .map(t =>

                typeof t === "object"

                    ? t.query

                    : t

            )

            .join(", ");

    const avgViews =

        channelProfile?.averageViews

        ||

        "N/A";

    const bestFormats =

        channelProfile?.bestFormats?.join(", ")

        ||

        "N/A";

    const worstFormats =

        channelProfile?.worstFormats?.join(", ")

        ||

        "N/A";

    return `

Sei l'intelligenza analitica di Brawl Analytics.

Non devi dare punteggi.

Devi solo restituire feature qualitative.

=== CHANNEL PROFILE ===

Average Views:

${avgViews}

Best Formats:

${bestFormats}

Worst Formats:

${worstFormats}

=== VIDEO ===

Description:

${proposal?.description || ""}

Idea originality:

${proposal?.ideaOriginality || ""}

Video originality:

${proposal?.videoOriginality || ""}

Format:

${proposal?.format || ""}

=== TRENDS ===

${listaTrends}

Rispondi SOLO con questo JSON:

{

"trendAlignment":"high/medium/low",

"semanticTrendSimilarity":"...",

"formatSuitability":"high/medium/low",

"historicalFit":"high/medium/low",

"innovation":"high/medium/low",

"competition":"high/medium/low",

"topicFreshness":"...",

"formatNovelty":"high/medium/low"

}

`;

}

async function callGroq(

    prompt,

    env

) {

    const response =

        await fetch(

            "https://api.groq.com/openai/v1/chat/completions",

            {

                method:

                    "POST",

                headers: {

                    Authorization:

                        `Bearer ${env.CHIAVE_API_AI}`,

                    "Content-Type":

                        "application/json"

                },

                body:

                    JSON.stringify({

                        model:

                            "llama-3.3-70b-versatile",

                        temperature:

                            0.2,

                        response_format: {

                            type:

                                "json_object"

                        },

                        messages: [

                            {

                                role:

                                    "system",

                                content:

                                    "Reply ONLY with valid JSON."

                            },

                            {

                                role:

                                    "user",

                                content:

                                    prompt

                            }

                        ]

                    })

            }

        );

    if (!response.ok) {

        throw new Error(

            await response.text()

        );

    }

    const result =

        await response.json();

    return JSON.parse(

        result

            .choices[0]

            .message.content

    );

}

export default {

    async fetch(

        request,

        env

    ) {

        const origin =

            request.headers.get(

                "Origin"

            );

        if (

            request.method ===

            "OPTIONS"

        ) {

            return new Response(

                null,

                {

                    headers:

                        corsHeaders(origin)

                }

            );

        }

        if (

            origin !== ALLOWED_ORIGIN

        ) {

            return json(

                {

                    success:

                        false,

                    error:

                        "Forbidden origin"

                },

                403,

                origin

            );

        }

        const url =

            new URL(

                request.url

            );

        if (

            request.method !== "POST"

        ) {

            return json(

                {

                    success:

                        false,

                    error:

                        "Method not allowed"

                },

                405,

                origin

            );

        }

        if (

            url.pathname !== "/analyze"

        ) {

            return json(

                {

                    success:

                        false,

                    error:

                        "Route not found"

                },

                404,

                origin

            );

        }

        try {

            const body =

                await request.json();

            if (!body.proposal) {

                return json(

                    {

                        success:

                            false,

                        error:

                            "Missing proposal"

                    },

                    400,

                    origin

                );

            }

            const trends =

                await getDailyTrends(

                    env

                );

            const prompt =

                buildPrompt(

                    body.proposal,

                    body.channelProfile,

                    trends

                );

            const aiFeatures =

                await callGroq(

                    prompt,

                    env

                );

            return json(

                {

                    success:

                        true,

                    aiFeatures,

                    metadata: {

                        model:

                            "llama-3.3-70b-versatile",

                        timestamp:

                            new Date()

                                .toISOString()

                    }

                },

                200,

                origin

            );

        }

        catch (err) {

            return json(

                {

                    success:

                        false,

                    error:

                        err.message

                },

                500,

                origin

            );

        }

    }

};
