const ALLOWED_ORIGINS = [
    "https://brawl-analytics-official.vercel.app",
    "http://localhost:5500",
    "http://127.0.0.1:5500"
];

function corsHeaders(origin) {
    const headers = {
        "Access-Control-Allow-Methods": "POST, OPTIONS",
        "Access-Control-Allow-Headers": "Authorization, Content-Type",
        "Access-Control-Max-Age": "86400",
        "Vary": "Origin"
    };
    if (ALLOWED_ORIGINS.includes(origin)) headers["Access-Control-Allow-Origin"] = origin;
    return headers;
}

async function isValidSupabaseUser(authHeader, env) {
    if (!env.SUPABASE_URL || !env.SUPABASE_ANON_KEY || !/^Bearer\s+\S+$/i.test(authHeader || "")) return false;
    try {
        const response = await fetch(`${env.SUPABASE_URL.replace(/\/$/, "")}/auth/v1/user`, {
            headers: { Authorization: authHeader, apikey: env.SUPABASE_ANON_KEY }
        });
        return response.ok;
    } catch {
        return false;
    }
}

function json(body, status = 200, origin = ALLOWED_ORIGINS) {

    return new Response(

        JSON.stringify(body, null, 2),

        {

            status,

            headers: {

                "Content-Type":
                    "application/json",

                ...corsHeaders(origin)

            }

        }

    );

}

async function getDailyTrends(env) {

    const fallbackTrends = [

        {

            query:
                "brawl stars meta tier list"

        },

        {

            query:
                "ranked mode global leaderboard"

        },

        {

            query:
                "best brawlers hypercharge"

        }

    ];

    try {

        const response =
            await fetch(

                "https://trends.angeskicollab10.workers.dev/trends"

            );

        if (!response.ok) {

            return {

                trends:
                    fallbackTrends

            };

        }

        const data =
            await response.json();

        if (

            !Array.isArray(data.trends)

            ||

            data.trends.length === 0

        ) {

            return {

                trends:
                    fallbackTrends

            };

        }

        return data;

    }

    catch {

        return {

            trends:
                fallbackTrends

        };

    }

}

function buildPrompt(

    proposal,

    channelProfile,

    trends

) {

    const listaTrends =

        (trends?.trends || [])

            .map(t =>

                typeof t === "object"

                    ? t.query

                    : t

            )

            .join(", ");

    const avgViews =

        channelProfile?.averageViews

        ||

        "N/A";

    const bestFormats =

        channelProfile?.bestFormats?.join(", ")

        ||

        "N/A";

    const worstFormats =

        channelProfile?.worstFormats?.join(", ")

        ||

        "N/A";

    return `

Sei l'intelligenza analitica di Brawl Analytics.

Non devi dare punteggi.

Devi solo restituire feature qualitative.

=== CHANNEL PROFILE ===

Average Views:

${avgViews}

Best Formats:

${bestFormats}

Worst Formats:

${worstFormats}

=== VIDEO ===

Description:

${proposal?.description || ""}

Idea originality:

${proposal?.ideaOriginality || ""}

Video originality:

${proposal?.videoOriginality || ""}

Format:

${proposal?.format || ""}

=== TRENDS ===

${listaTrends}

Rispondi SOLO con questo JSON:

{

"trendAlignment":"high/medium/low",

"semanticTrendSimilarity":"...",

"formatSuitability":"high/medium/low",

"historicalFit":"high/medium/low",

"innovation":"high/medium/low",

"competition":"high/medium/low",

"topicFreshness":"...",

"formatNovelty":"high/medium/low"

}

`;

}

async function callGroq(

    prompt,

    env

) {

    if (!env.GROQ_API_KEY || !env.GROQ_MODEL) {

        throw new Error("Missing GROQ_API_KEY or GROQ_MODEL");

    }

    const response =

        await fetch(

            "https://api.groq.com/openai/v1/chat/completions",

            {

                method:

                    "POST",

                headers: {

                    Authorization:

                        `Bearer ${env.GROQ_API_KEY}`,

                    "Content-Type":

                        "application/json"

                },

                body:

                    JSON.stringify({

                        model:

                            env.GROQ_MODEL,

                        temperature:

                            0.2,

                        response_format: {

                            type:

                                "json_object"

                        },

                        messages: [

                            {

                                role:

                                    "system",

                                content:

                                    "Reply ONLY with valid JSON."

                            },

                            {

                                role:

                                    "user",

                                content:

                                    prompt

                            }

                        ]

                    })

            }

        );

    if (!response.ok) {

        throw new Error(

            await response.text()

        );

    }

    const result =

        await response.json();

    return JSON.parse(

        result

            .choices[0]

            .message.content

    );

}

export default {

    async fetch(

        request,

        env

    ) {

        const origin =

            request.headers.get(

                "Origin"

            );

        if (

            request.method ===

            "OPTIONS"

        ) {

            return new Response(

                null,

                {

                    headers:

                        corsHeaders(origin)

                }

            );

        }

        const authHeader = request.headers.get("Authorization");
        if (!authHeader || !(await isValidSupabaseUser(authHeader, env))) {
            return json({ success: false, error: "Unauthorized" }, 401, origin);
        }

        if (!ALLOWED_ORIGINS.includes(origin)) {

            return json(

                {

                    success:

                        false,

                    error:

                        "Forbidden origin"

                },

                403,

                origin

            );

        }

        const url =

            new URL(

                request.url

            );

        if (

            request.method !== "POST"

        ) {

            return json(

                {

                    success:

                        false,

                    error:

                        "Method not allowed"

                },

                405,

                origin

            );

        }

        if (

            url.pathname !== "/analyze"

        ) {

            return json(

                {

                    success:

                        false,

                    error:

                        "Route not found"

                },

                404,

                origin

            );

        }

        try {

            const contentLength = Number(request.headers.get("Content-Length") || 0);
            if (contentLength > 64 * 1024) {
                return json({ success: false, error: "Request payload is too large" }, 413, origin);
            }

            const body =

                await request.json();

            if (!body.proposal || typeof body.proposal !== "object") {

                return json(

                    {

                        success:

                            false,

                        error:

                            "Missing proposal"

                    },

                    400,

                    origin

                );

            }

            const trends =

                await getDailyTrends(

                    env

                );

            const prompt =

                buildPrompt(

                    body.proposal,

                    body.channelProfile,

                    trends

                );

            const aiFeatures =

                await callGroq(

                    prompt,

                    env

                );

            return json(

                {

                    success:

                        true,

                    aiFeatures,

                    metadata: {

                        model:

                            env.GROQ_MODEL,

                        timestamp:

                            new Date()

                                .toISOString()

                    }

                },

                200,

                origin

            );

        }

        catch (err) {
            console.error("Virality worker request failed:", err);
            return json(
                { success: false, error: "AI service temporarily unavailable" },
                502,
                origin
            );
        }

    }

};
