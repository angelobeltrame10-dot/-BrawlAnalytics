const ALLOWED_ORIGINS = [
    "https://brawl-analytics-official.vercel.app",
    "http://localhost:5500",
    "http://127.0.0.1:5500"
];

function corsHeaders(origin = "") {
    const headers = {
        "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
        "Access-Control-Allow-Headers": "Content-Type, Authorization",
        "Access-Control-Max-Age": "86400",
        "Vary": "Origin"
    };

    if (ALLOWED_ORIGINS.includes(origin)) {
        headers["Access-Control-Allow-Origin"] = origin;
    }

    return headers;
}

function jsonResponse(body, status = 200, origin = "") {
    return Response.json(body, { status, headers: corsHeaders(origin) });
}

export default {
    async fetch(request, env) {
        const origin = request.headers.get("Origin") || "";
        const respond = (body, status = 200) => jsonResponse(body, status, origin);

        if (request.method === "OPTIONS") {
            return new Response(null, { status: 204, headers: corsHeaders(origin) });
        }

        // ============================================================
        // GET — lettura pubblica usata dalla dashboard.
        // Nessun Authorization richiesto: sono trend aggregati, non
        // dati sensibili, e il MAKE_SECRET non deve mai finire nel
        // browser dell'utente.
        // ============================================================
        if (request.method === "GET") {

            try {
                const stored = await env.BRAWL_KV.get("latest_creator_trends");

                if (!stored) {
                    return respond({ success: false, error: "No creator trends available yet" }, 404);
                }

                return new Response(stored, {
                    status: 200,
                    headers: { "Content-Type": "application/json", ...corsHeaders(origin) }
                });

            } catch (err) {
                return respond({ success: false, error: err.message }, 500);
            }

        }

        // ============================================================
        // POST — scrittura riservata a Make (Bearer token richiesto)
        // ============================================================
        if (request.method !== "POST") {
            return respond({ success: false, error: "Method Not Allowed" }, 405);
        }

        const authHeader = request.headers.get("Authorization");

        if (!authHeader || authHeader !== `Bearer ${env.MAKE_SECRET}`) {
            return respond({ success: false, error: "Unauthorized" }, 401);
        }

        try {

            const data = await request.json();

            if (!data || typeof data !== "object" || Array.isArray(data)) {
                return respond({ success: false, error: "Invalid JSON payload" }, 400);
            }

            // Valida che il payload non sia "vuoto" prima di sovrascrivere il KV.
            // Se la generazione lato Make è fallita silenziosamente pur restituendo
            // un payload strutturalmente valido, non vogliamo sovrascrivere l'ultimo
            // risultato buono con uno vuoto.
            const trendsData = data.trends && data.trends[0] ? data.trends[0] : null;
            const hasValidData = trendsData && (
                (trendsData.videosAnalyzed && trendsData.videosAnalyzed > 0) ||
                (Array.isArray(trendsData.formats) && trendsData.formats.length > 0) ||
                (Array.isArray(trendsData.topics) && trendsData.topics.length > 0) ||
                (Array.isArray(trendsData.brawlers) && trendsData.brawlers.length > 0) ||
                (Array.isArray(trendsData.keywords) && trendsData.keywords.length > 0) ||
                (Array.isArray(trendsData.trends) && trendsData.trends.length > 0)
            );

            if (!hasValidData) {
                console.log("Empty analysis payload rejected, previous data retained", data);
                return respond({ success: false, error: "Empty analysis payload rejected, previous data retained" }, 400);
            }

            data.receivedAt = new Date().toISOString();

            await env.BRAWL_KV.put("latest_creator_trends", JSON.stringify(data));

            return respond({ success: true, message: "Creator trends updated." });

        } catch (err) {
            return respond({ success: false, error: err.message }, 500);
        }

    }
};
